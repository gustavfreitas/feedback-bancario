docker exec -it -u postgres postgres-db psql -d postgres -c "\copy (SELECT m.id AS manifestation_id, m.feedback_text, m.satisfaction_score, m.sentiment, m.urgency, m.created_at, c.name AS channel_name, mt.name AS type_name, ps.name AS product_or_service_name FROM manifestations m LEFT JOIN channels c ON m.channel_id = c.id LEFT JOIN manifestation_types mt ON m.manifestation_type_id = mt.id LEFT JOIN manifestation_products mp ON m.id = mp.manifestation_id LEFT JOIN products_services ps ON mp.product_id = ps.id) TO STDOUT WITH CSV HEADER" > relatorio_manifestacoes.csv
