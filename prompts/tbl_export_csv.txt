docker exec -u postgres postgres-db psql -d postgres -c "SET client_encoding='UTF8'; COPY (SELECT m.id AS manifestation_id, m.feedback_text, m.satisfaction_score, m.sentiment, m.urgency, m.created_at, c.name AS channel_name, mt.name AS type_name, ps.name AS product_or_service_name FROM manifestations m LEFT JOIN channels c ON m.channel_id = c.id LEFT JOIN manifestation_types mt ON m.manifestation_type_id = mt.id LEFT JOIN manifestation_products mp ON m.id = mp.manifestation_id LEFT JOIN products_services ps ON mp.product_id = ps.id ORDER BY m.id) TO '/tmp/relatorio_manifestacoes.csv' WITH (FORMAT csv, HEADER true, ENCODING 'UTF8')"
docker cp postgres-db:/tmp/relatorio_manifestacoes.csv ./relatorio_manifestacoes.csv
